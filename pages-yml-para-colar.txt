    label: Páginas da wiki
    type: collection
    path: conteudo/paginas
    format: yaml-frontmatter

    view:
      fields: [titulo, grupo, subtitulo]
      primary: titulo
      sort: [ordem, titulo]
      search: [titulo, subtitulo]
    fields:

        type: string

      - name: lado

        type: select
        description: Em que bloco o personagem aparece na aba Personagens. Em páginas de grupo, use "Personagens" para listar o grupo junto com os personagens.
        options:
          values: [Ordem do Fogo, Aliados, Domínio Sombrio, Entidades, Outros, Personagens]
      - name: tipo_criatura
        label: Tipo de criatura (só criaturas)
        type: select

      - name: subtitulo

          values:

              label: Obra oficial
            - name: discord
              label: Falas do Toom no Discord
      - name: estreia_novel
        label: Aparece pela primeira vez no capítulo da novel
        type: number
        description: 'Para o filtro "Li até": quem leu menos que isso vê um aviso de spoiler antes da página. Se ficar vazio, o site usa o primeiro {cap N} do texto.'
      - name: ordem
        label: Ordem na lista
        type: number
        description: Números menores aparecem primeiro dentro da aba.
      - name: imagem_card
        label: Imagem do cartão
        type: image
        description: Aparece atrás do nome nas listas das abas, escurecida embaixo para o texto ler bem. Use uma imagem deitada (retangular). Se ficar vazio, o site usa a primeira de "Outras versões" ou a imagem principal.

        label: Imagem principal
        type: image
      - name: foto_legenda
        label: Legenda da imagem principal
        type: string
      - name: ficha_nome
        label: Nome na ficha
        type: string
        description: Nome completo que aparece no topo da ficha lateral.
      - name: ficha
        label: Ficha lateral
        type: object
        list: true
        description: 'Cada linha da ficha. No valor dá para usar links, ex.: [Ordem do Fogo](#ordem-do-fogo).'
        fields:
          - name: campo
            label: Campo
            type: string
          - name: valor
            label: Valor
            type: string
      - name: body
        label: Texto da página
        type: rich-text
        options:
          format: markdown
          switcher: true
          media: false
        description: 'Para citar um capítulo da novel, escreva {cap 3}. Para um extra da HQ, {hq 3}. Para linkar outra página, use o link #nome-da-pagina (ex.: #any).'
      - name: galeria
        label: Galeria
        type: object
        list: true
        fields:
          - name: imagem
            type: image
          - name: legenda
            type: string
      - name: versoes
        label: Outras versões (artes)
        type: object
        list: true
        fields:
          - name: imagem
            type: image
          - name: legenda
            type: string
      - name: falas_marcantes
        label: Falas marcantes (balões)
        type: object
        list: true
        fields:
          - name: fala
            label: Fala
            type: text
          - name: contexto
            label: Quando foi dita
            type: string
          - name: capitulo
            label: Capítulo da novel
            type: number
      - name: extras_do_autor
        label: Extras do autor (fim da HQ)
        type: object
        list: true
        fields:
          - name: texto
            type: text
          - name: hq
            label: Capítulo da HQ
            type: number
      - name: titulo_falas
        label: Título da seção do Discord
        type: string
        description: Deixe vazio para usar "Segundo o Toom (Discord)".
      - name: secoes_como_titulos
        label: Mostrar cada seção do Discord como título grande
        type: boolean
      - name: segundo_o_toom
        label: Segundo o Toom (falas do Discord)
        type: object
        list: true
        fields:
          - name: texto
            label: O que ele disse
            type: text
          - name: secao
            label: Seção
            type: string
            description: 'Em personagens: Aparência, Poderes e luta, Gostos e personalidade, Família e passado ou Na história e outros. Pode deixar vazio.'
          - name: data
            label: Data (AAAA-MM-DD)
            type: string
            pattern: "^(\\d{4}-\\d{2}-\\d{2})?$"
          - name: link
            label: Link da mensagem no Discord
            type: string
          - name: destaque
            label: Mostrar também em Curiosidades
            type: boolean
